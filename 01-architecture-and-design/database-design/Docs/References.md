## SECTION 7: CHEAT SHEETS

### 7.1 PostgreSQL Cheat Sheet

```sql
-- CONNECTION
psql -U username -d database -h host -p port

-- BASIC COMMANDS
\l              -- List databases
\c database     -- Connect to database
\dt             -- List tables
\d table        -- Describe table
\di             -- List indexes
\df             -- List functions
\q              -- Quit

-- SQL QUICK REFERENCE
SELECT * FROM table;
INSERT INTO table (col1, col2) VALUES (val1, val2);
UPDATE table SET col1 = val1 WHERE condition;
DELETE FROM table WHERE condition;

-- INDEXING
CREATE INDEX idx_name ON table(column);
CREATE INDEX CONCURRENTLY idx_name ON table(column);
CREATE INDEX idx_name ON table(column) WHERE condition;
CREATE INDEX idx_name ON table(col1, col2);

-- EXPLAIN
EXPLAIN SELECT * FROM table WHERE condition;
EXPLAIN ANALYZE SELECT * FROM table WHERE condition;
EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) SELECT * FROM table;

-- TRANSACTIONS
BEGIN;
-- operations
COMMIT;
ROLLBACK;
```

### 7.2 MongoDB Cheat Sheet

```javascript
// CONNECTION
mongosh -u username -p password

// BASIC COMMANDS
show dbs                    // List databases
use database               // Switch database
show collections           // List collections

// CRUD
db.collection.find()        // Read
db.collection.findOne()     // Read one
db.collection.insertOne()   // Create
db.collection.updateOne()   // Update
db.collection.deleteOne()   // Delete

// QUERIES
db.collection.find({ field: value })
db.collection.find({ field: { $gt: 100 } })
db.collection.find().sort({ field: -1 })
db.collection.find().limit(10)

// INDEXES
db.collection.createIndex({ field: 1 })
db.collection.getIndexes()
db.collection.dropIndex("field_1")

// AGGREGATION
db.collection.aggregate([
    { $match: { field: value } },
    { $group: { _id: "$field", count: { $sum: 1 } } },
    { $sort: { count: -1 } }
])
```

### 7.4 Neo4j Cheat Sheet

```cypher
// CONNECTION
cypher-shell -u username -p password

// CREATE NODES
CREATE (n:Label {property: value})

// CREATE RELATIONSHIPS
MATCH (a:Label), (b:Label)
CREATE (a)-[:RELATIONSHIP]->(b)

// QUERIES
MATCH (n:Label) RETURN n
MATCH (n:Label {property: value}) RETURN n
MATCH (n)-[r:RELATIONSHIP]->(m) RETURN n, r, m

// PATTERNS
MATCH (n)-[:RELATIONSHIP]->(m) WHERE n.property = value
MATCH path = (n)-[:RELATIONSHIP*1..3]->(m)

// AGGREGATION
MATCH (n:Label)-[:RELATIONSHIP]->(m)
RETURN n.property, COUNT(*)

// DELETE
MATCH (n) DETACH DELETE n
```
