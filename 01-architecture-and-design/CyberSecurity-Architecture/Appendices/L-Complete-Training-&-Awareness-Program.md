#### Practical Exercises

```python
# Secure Coding Exercise
# Implement secure API endpoint

from flask import Flask, request, jsonify
import re

app = Flask(__name__)

# ❌ Vulnerable implementation
@app.route('/api/search-bad', methods=['POST'])
def search_bad():
    query = request.json.get('query')
    # SQL injection vulnerability
    result = execute_query(f"SELECT * FROM users WHERE name = '{query}'")
    return jsonify(result)

# ✅ Secure implementation
@app.route('/api/search-good', methods=['POST'])
def search_good():
    query = request.json.get('query')
    # Input validation
    if not re.match('^[a-zA-Z0-9 ]+$', query):
        return jsonify({"error": "Invalid input"}), 400
    # Parameterized query
    result = execute_query("SELECT * FROM users WHERE name = %s", (query,))
    return jsonify(result)
```
