## Target 7.1: Configuring Media Files for Uploads

### The Concept

**Media files** are user-uploaded content (images, documents, etc.). They're different from static files (CSS, JS) because they're generated by users and need to be stored persistently.

Think of it like this:
- **Static files**: Store-bought furniture (CSS, JS, images you provide)
- **Media files**: User's personal photos (uploaded by users)

### The Implementation

```python
"""
Script to generate default avatar images.
Run this once to create the default avatar file.
"""

import os
from PIL import Image, ImageDraw, ImageFont
from django.conf import settings
from pathlib import Path

def generate_default_avatar():
    """Generate a default avatar image."""
    # Create a 200x200 image with a gradient background
    size = (200, 200)
    image = Image.new('RGB', size, color=(52, 152, 219))  # Django blue
    
    # Draw a circle
    draw = ImageDraw.Draw(image)
    draw.ellipse([(0, 0), size], fill=(41, 128, 185))
    
    # Add a user icon (simple stick figure or initials)
    # For simplicity, we'll add text "D" for Django
    try:
        # Try to use a system font
        font = ImageFont.truetype("/usr/share/fonts/truetype/liberation/LiberationSans-Bold.ttf", 80)
    except:
        # Fallback to default
        font = ImageFont.load_default()
    
    # Draw "D" in the center
    draw.text((80, 50), "D", fill=(255, 255, 255), font=font)
    
    # Ensure the directory exists
    avatar_dir = Path(settings.STATIC_ROOT) / 'blog' / 'images'
    avatar_dir.mkdir(parents=True, exist_ok=True)
    
    # Save the image
    avatar_path = avatar_dir / 'default-avatar.png'
    image.save(avatar_path)
    print(f"Default avatar created at: {avatar_path}")

if __name__ == "__main__":
    generate_default_avatar()
```

## Target 7.3: Implementing Advanced File Upload Validation

### The Concept

When users upload files, we need to validate them for security and performance. We'll add:
- File type checking
- File size limits
- Image dimension validation
- Secure file naming

### The Implementation

**File: `blog/validators.py`** (create new file)

```python
from django.core.exceptions import ValidationError
from django.utils.translation import gettext_lazy as _
from django.conf import settings
from PIL import Image
import os
import magic  # Python library for file type detection


def validate_file_size(value, max_size=settings.MAX_IMAGE_SIZE):
    """
    Validate that a file doesn't exceed the maximum size.
    
    Args:
        value: The uploaded file
        max_size: Maximum file size in bytes
        
    Raises:
        ValidationError: If file is too large
    """
    if value.size > max_size:
        raise ValidationError(
            _('File size must be under %(max_size)s MB.'),
            params={'max_size': max_size / (1024 * 1024)}
        )


def validate_image_type(value):
    """
    Validate that a file is an allowed image type.
    
    Args:
        value: The uploaded file
        
    Raises:
        ValidationError: If file type is not allowed
    """
    # Check MIME type
    mime_type = magic.from_buffer(value.read(1024), mime=True)
    value.seek(0)  # Reset file pointer
    
    if mime_type not in settings.ALLOWED_IMAGE_TYPES:
        raise ValidationError(
            _('Only JPEG, PNG, GIF, and WebP images are allowed.')
        )


def validate_image_dimensions(value, min_width=100, min_height=100, 
                              max_width=2000, max_height=2000):
    """
    Validate image dimensions.
    
    Args:
        value: The uploaded file
        min_width: Minimum width in pixels
        min_height: Minimum height in pixels
        max_width: Maximum width in pixels
        max_height: Maximum height in pixels
        
    Raises:
        ValidationError: If dimensions are outside allowed range
    """
    try:
        img = Image.open(value)
        width, height = img.size
        
        if width < min_width or height < min_height:
            raise ValidationError(
                _('Image must be at least %(min_width)sx%(min_height)s pixels.'),
                params={'min_width': min_width, 'min_height': min_height}
            )
        
        if width > max_width or height > max_height:
            raise ValidationError(
                _('Image must be at most %(max_width)sx%(max_height)s pixels.'),
                params={'max_width': max_width, 'max_height': max_height}
            )
    except Exception as e:
        raise ValidationError(_('Unable to process image: %(error)s'), 
                            params={'error': str(e)})


def validate_filename_safety(value):
    """
    Validate that a filename contains only safe characters.
    
    Args:
        value: The uploaded file
        
    Raises:
        ValidationError: If filename contains unsafe characters
    """
    import re
    
    filename = value.name
    # Remove path separators
    if '/' in filename or '\\' in filename:
        raise ValidationError(_('Filename cannot contain path separators.'))
    
    # Allow only alphanumeric, dash, underscore, and dot
    if not re.match(r'^[a-zA-Z0-9_.-]+$', filename):
        raise ValidationError(_('Filename contains invalid characters.'))
```

## Target 7.5: Working with Sessions

### The Concept

**Sessions** allow you to store temporary data about a user between requests. This is useful for:
- Shopping carts
- Recently viewed items
- Temporary preferences
- Form wizard data

### The Implementation

We'll add a "recently viewed posts" feature using sessions:

**File: `blog/views.py`** (update PostDetailView)

```python
# ... keep existing imports ...

class PostDetailView(DetailView):
    # ... keep existing attributes ...
    
    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        post = self.get_object()
        
        # ... existing context setup ...
        
        # Track recently viewed posts in session
        recent_posts = self.request.session.get('recent_posts', [])
        
        # Add current post to recent posts (if published)
        if post.status == Post.Status.PUBLISHED:
            # Remove if already exists (to move to top)
            if post.id in recent_posts:
                recent_posts.remove(post.id)
            # Add to front
            recent_posts.insert(0, post.id)
            # Keep only last 5
            recent_posts = recent_posts[:5]
            
            # Save back to session
            self.request.session['recent_posts'] = recent_posts
        
        # Get the actual post objects for display
        recent_post_objects = []
        for post_id in recent_posts[:5]:
            try:
                recent_post = Post.objects.get(id=post_id, status=Post.Status.PUBLISHED)
                recent_post_objects.append(recent_post)
            except Post.DoesNotExist:
                pass
        
        context['recently_viewed'] = recent_post_objects
        
        return context
```

## Target 7.6: Implementing Database Transactions

### The Concept

**Transactions** ensure that a group of database operations either all succeed or all fail. This prevents data inconsistency.

Common use cases:
- Creating a post with tags
- Transferring data between models
- Batch updates that must be atomic

### The Implementation

**File: `blog/services/post_service.py`** (update with transaction)

```python
from django.db import transaction
from django.core.exceptions import ValidationError
from .models import Post, Tag


class PostService:
    # ... existing methods ...
    
    @staticmethod
    @transaction.atomic
    def create_post_with_tags(data, author):
        """
        Create a post and its tags in a single transaction.
        
        If anything fails, the entire operation rolls back.
        """
        # Validate and create the post
        post = PostService.create_post(data, author)
        
        # Process tags inside the same transaction
        tags_input = data.get('tags_input', '')
        if tags_input:
            tag_names = [tag.strip().lower() for tag in tags_input.split(',') if tag.strip()]
            
            for tag_name in tag_names:
                tag, created = Tag.objects.get_or_create(
                    name=tag_name,
                    defaults={'slug': slugify(tag_name)}
                )
                post.tags.add(tag)
        
        return post
    
    @staticmethod
    @transaction.atomic
    def bulk_update_posts(post_ids, update_data):
        """
        Update multiple posts at once.
        
        This ensures all updates succeed or fail together.
        """
        if not post_ids:
            return
        
        # Get all posts to update
        posts = Post.objects.filter(id__in=post_ids)
        
        # Update each post
        for post in posts:
            for key, value in update_data.items():
                setattr(post, key, value)
            post.save()
        
        return posts.count()
    
    @staticmethod
    @transaction.atomic
    def move_posts_to_category(post_ids, category):
        """
        Move multiple posts to a new category.
        """
        if not post_ids:
            return
        
        posts = Post.objects.filter(id__in=post_ids)
        updated = posts.update(category=category)
        
        return updated
```

```python
from django.core.management.base import BaseCommand
from django.utils import timezone
from blog.models import Post


class Command(BaseCommand):
    """
    Management command to publish scheduled posts.
    
    Usage: python manage.py publish_scheduled_posts
    """
    help = 'Publish posts that are scheduled for publication'
    
    def handle(self, *args, **options):
        now = timezone.now()
        
        # Find draft posts with a scheduled publish date in the past
        scheduled_posts = Post.objects.filter(
            status=Post.Status.DRAFT,
            scheduled_publish_at__lte=now
        )
        
        count = scheduled_posts.count()
        
        if count == 0:
            self.stdout.write('No posts scheduled for publication.')
            return
        
        for post in scheduled_posts:
            post.status = Post.Status.PUBLISHED
            post.published_at = now
            post.save()
            self.stdout.write(f'Published: {post.title}')
        
        self.stdout.write(
            self.style.SUCCESS(f'Successfully published {count} post(s).')
        )
```
